\begin{definition}[Lipschitz continuity]
A function $f\colon X\to Y$ between metric spaces is \emph{Lipschitz
continuous} if there exists $L\geq 0$ such that
\[
  d_Y\bigl(f(x),f(y)\bigr)\leq Ld_X(x,y)
\]
for every $x,y\in X$.
\end{definition}

\begin{theorem}[Lipschitz functions are uniformly continuous]
Every Lipschitz continuous function is uniformly continuous.
\end{theorem}

\begin{proof}
Let $L$ be a Lipschitz constant. If $L=0$, the function is constant. Otherwise,
given $\varepsilon>0$, choose $\delta=\varepsilon/L$. Then
$d_X(x,y)<\delta$ implies
$d_Y(f(x),f(y))\leq Ld_X(x,y)<\varepsilon$.
\end{proof}

\begin{eg}[A concrete function]
The function $f(x)=x^2$ is Lipschitz continuous on $[0,1]$, since
$|x^2-y^2|=|x-y||x+y|\leq 2|x-y|$.
\end{eg}

\begin{prop}[Sums of Lipschitz functions]
If $f$ and $g$ have Lipschitz constants $L_f$ and $L_g$, respectively, then
$f+g$ has Lipschitz constant $L_f+L_g$.
\end{prop}

\begin{explanation}
The Lipschitz constant is a global bound on steepness. Adding two functions
adds their possible changes, which explains why their constants add.
\end{explanation}

\begin{remark}
Uniform continuity does not imply Lipschitz continuity in general; for example,
$f(x)=\sqrt{x}$ on $[0,1]$ is uniformly continuous but not Lipschitz.
\end{remark}

\subsection{Exercises}

\begin{exercise}
Show that $f(x)=3x-2$ is Lipschitz on $\R$ and find a Lipschitz constant.
\end{exercise}

\begin{solution}
$|f(x)-f(y)|=|3x-3y|=3|x-y|$, so $L=3$ is a Lipschitz constant.
\end{solution}
